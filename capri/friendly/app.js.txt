const $ = (id) => document.getElementById(id);
const pages = ["welcome", "connection", "settings", "chat", "closed"];
const emptyChat = $("empty-chat").cloneNode(true);
let state = { preferences: { name: "", tone: "natural", textSize: "normal", contrast: false }, providers: [], models: [], sessions: [] };
let current = null, messages = [], responding = false, sending = false, refreshing = false, waiting = false, view = "welcome", messageSignature = "", closed = false;
let allProviders = false;
const storageKey = "capri-conexion-" + location.port;
let token = location.hash.slice(1);
try { if (token) sessionStorage.setItem(storageKey, token); else token = sessionStorage.getItem(storageKey) || ""; } catch {}
history.replaceState(null, "", location.pathname);
const keyPages = { openrouter: "https://openrouter.ai/keys", groq: "https://console.groq.com/keys", google: "https://aistudio.google.com/apikey", anthropic: "https://console.anthropic.com/settings/keys", openai: "https://platform.openai.com/api-keys" };

async function api(route, method = "GET", body) {
  const response = await fetch("/friendly/" + route, { method, headers: { "X-Capri-Token": token, ...(body === undefined ? {} : { "Content-Type": "application/json" }) }, ...(body === undefined ? {} : { body: JSON.stringify(body) }) });
  const result = await response.json();
  if (!response.ok) throw new Error(result.error || "La conexión no está disponible. Abre Capri desde su acceso directo.");
  return result;
}
function notice(message) { $("notice").textContent = message; $("notice").hidden = !message; }
function element(tag, className, text) { const node = document.createElement(tag); if (className) node.className = className; if (text !== undefined) node.textContent = text; return node; }
function ready() { return state.models.some((model) => model.id === state.preferences.model?.id && model.providerID === state.preferences.model?.providerID); }
function applyPreferences() { document.body.classList.toggle("large-text", state.preferences.textSize === "grande"); document.body.classList.toggle("high-contrast", state.preferences.contrast); }
function modelKey(model) { return model.providerID + "\u001f" + model.id; }
function show(next, focus = true) {
  if (closed && next !== "closed") return;
  view = next;
  document.body.dataset.view = next;
  for (const page of pages) $(page).hidden = page !== next;
  $("notice").hidden = true;
  if (next === "settings") {
    $("preferred-name").value = state.preferences.name;
    $("tone").value = state.preferences.tone;
    $("text-size").value = state.preferences.textSize;
    $("contrast").checked = state.preferences.contrast;
  }
  if (next === "chat") updateControls();
  if (focus) { const heading = $(next).querySelector("h1"); if (heading) { heading.tabIndex = -1; heading.focus({ preventScroll: true }); } }
}
function updateControls() {
  const busy = responding || sending;
  $("new-chat").disabled = busy;
  $("send").disabled = busy || !ready() || !$("message").value.trim();
  $("send").hidden = responding;
  $("stop").hidden = !responding;
  $("message").disabled = !ready() || sending;
  $("needs-connection").hidden = ready();
  if (responding) $("reply-status").textContent = "Capri está respondiendo…";
  else if ($("reply-status").textContent === "Capri está respondiendo…") $("reply-status").textContent = "";
  $("export").hidden = !messages.length;
  for (const button of $("history").querySelectorAll("button")) button.disabled = busy;
}
function renderHistory() {
  $("history").replaceChildren();
  $("history-count").textContent = state.sessions.length ? String(state.sessions.length) : "";
  if (!state.sessions.length) $("history").append(element("p", "muted small", "Tus charlas aparecerán aquí."));
  for (const session of state.sessions) {
    const row = element("div", "history-row" + (session.id === current ? " active" : ""));
    const button = element("button", "history-select", session.title);
    button.title = session.title;
    if (session.id === current) button.setAttribute("aria-current", "true");
    button.addEventListener("click", () => run(() => selectChat(session.id)));
    const archive = element("button", "history-archive", "×");
    archive.setAttribute("aria-label", "Archivar conversación: " + session.title);
    archive.addEventListener("click", () => confirmAction("¿Archivar esta conversación?", "Dejará de aparecer en la lista. Su contenido se conservará en el perfil local.", "Archivar", async () => {
      await api("chat/" + session.id + "/archive", "POST", {});
      if (current === session.id) resetChat();
      await loadState(); notice("Conversación archivada.");
    }));
    row.append(button, archive); $("history").append(row);
  }
  updateControls();
}
function renderConnection() {
  const previousProvider = $("provider").value;
  $("provider").replaceChildren(new Option("Selecciona tu servicio", ""));
  const familiar = new Set(["openai", "anthropic", "google", "openrouter", "groq"]);
  const providers = allProviders ? state.providers : state.providers.filter((item) => familiar.has(item.id) || item.connected);
  $("other-providers").hidden = providers.length === state.providers.length;
  for (const provider of [...providers].sort((a, b) => Number(b.connected) - Number(a.connected) || a.name.localeCompare(b.name, "es")))
    $("provider").add(new Option(provider.name + (provider.connected ? " · conectado" : ""), provider.id));
  if (state.providers.some((item) => item.id === previousProvider)) $("provider").value = previousProvider;
  else if (state.providers.some((item) => item.connected)) $("provider").value = state.providers.find((item) => item.connected).id;
  const chosen = state.preferences.model ? modelKey(state.preferences.model) : $("model").value;
  $("model").replaceChildren();
  for (const model of state.models) $("model").add(new Option(model.name + " · " + (state.providers.find((p) => p.id === model.providerID)?.name || model.providerID), modelKey(model)));
  if (state.models.some((model) => modelKey(model) === chosen)) $("model").value = chosen;
  if (!state.models.length) $("model").add(new Option("La lista de modelos todavía no está disponible", ""));
  $("model-form").hidden = !state.providers.some((item) => item.connected);
  $("model-form").querySelector("button[type=submit]").disabled = !state.models.length;
  $("connection-label").textContent = ready() ? "IA configurada" : "Conectar IA";
  $("connection-open").classList.toggle("connected", ready());
  providerLink(); updateControls();
}
function providerLink() { const url = keyPages[$("provider").value]; $("key-link").hidden = !url; if (url) $("key-link").href = url; }
async function loadState() { state = await api("state"); applyPreferences(); renderHistory(); renderConnection(); }
function resetChat() { current = null; messages = []; messageSignature = ""; waiting = false; responding = false; $("chat-title").textContent = "Tu espacio con Capri."; $("messages").replaceChildren(emptyChat.cloneNode(true)); $("chat-error").hidden = true; $("message").value = ""; renderHistory(); show("chat"); }
async function selectChat(id) { current = id; waiting = false; messageSignature = ""; show("chat"); await refreshChat(); renderHistory(); $("history").classList.remove("mobile-open"); $("history-toggle").setAttribute("aria-expanded", "false"); }
function renderMessages(next) {
  const container = $("messages");
  const nearBottom = container.scrollHeight - container.clientHeight - container.scrollTop < 100;
  container.replaceChildren();
  for (const message of next) {
    const article = element("article", "message " + message.role);
    article.append(element("p", "message-name", message.role === "user" ? (state.preferences.name || "TÚ") : "CAPRI"));
    article.append(element("div", "message-body", message.text));
    if (message.error) article.append(element("p", "message-error", "El servicio de IA no pudo terminar esta respuesta. Revisa tu conexión y sus límites antes de volver a enviar un mensaje."));
    container.append(article);
  }
  if (!next.length) container.append(emptyChat.cloneNode(true));
  if (nearBottom || waiting) container.scrollTop = container.scrollHeight;
}
async function refreshChat() {
  if (!current || refreshing || closed) return;
  refreshing = true;
  const requested = current;
  try {
    const snapshot = await api("chat/" + requested);
    if (current !== requested) return;
    const wasResponding = responding;
    messages = snapshot.messages;
    const last = messages.at(-1);
    if (last?.role === "assistant" && !snapshot.active) waiting = false;
    if (snapshot.outcome === "failed" && !snapshot.active) waiting = false;
    responding = snapshot.active || waiting;
    $("chat-title").textContent = snapshot.title === "Nueva conversación" ? "Tu espacio con Capri." : snapshot.title || "Conversación";
    const signature = JSON.stringify(messages);
    if (signature !== messageSignature) { renderMessages(messages); messageSignature = signature; }
    $("chat-error").hidden = true;
    updateControls();
    if (wasResponding && !responding) { await loadState(); $("reply-status").textContent = last?.error || snapshot.outcome === "failed" ? "La respuesta no se completó. Revisa Conexión." : "Capri terminó de responder."; }
  } catch (error) { $("chat-error").textContent = error.message + " Volveremos a comprobar la conexión."; $("chat-error").hidden = false; }
  finally { refreshing = false; }
}
function run(action) { Promise.resolve().then(action).catch((error) => notice(error.message || "No se pudo completar la acción. Inténtalo de nuevo.")); }
let pendingConfirm = null;

$("confirm-dialog").addEventListener("cancel", () => { pendingConfirm = null; });

      await api("connect", "POST", { provider: $("provider").value, key: $("api-key").value });

    finally { $("connect-submit").disabled = false; }
  })();

$("chat-form").addEventListener("submit", (event) => {
  event.preventDefault(); if (sending || responding || !$("message").value.trim()) return;
  const draft = $("message").value.trim(); sending = true; notice(""); $("chat-error").hidden = true; updateControls();
  (async () => {
    try {
      if (!current) { current = (await api("chat", "POST", {})).id; messageSignature = ""; }
      await api("chat/" + current + "/message", "POST", { text: draft });
      $("message").value = ""; waiting = true; responding = true;
      await refreshChat(); await loadState();
    } catch (error) { $("chat-error").textContent = error.message; $("chat-error").hidden = false; }
    finally { sending = false; updateControls(); $("message").focus(); }
  })();
});

  const content = "Conversación con Capri\n" + new Date().toLocaleString("es-MX") + "\n\n" + messages.map((message) => (message.role === "user" ? state.preferences.name || "Tú" : "Capri") + ":\n" + message.text).join("\n\n");
  const url = URL.createObjectURL(new Blob([content], { type: "text/plain;charset=utf-8" })); const link = element("a"); link.href = url; link.download = "Conversacion-con-Capri.txt"; link.click(); setTimeout(() => URL.revokeObjectURL(url), 1000);
});
